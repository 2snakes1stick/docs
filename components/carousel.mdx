# Carousel Component

A highly customizable and accessible carousel component designed for showcasing images, content cards, and various media types. Built with performance and user experience in mind.

## Overview

The Carousel component provides:
- **Responsive Design**: Adapts to different screen sizes and orientations
- **Touch Gestures**: Full touch and swipe support for mobile devices
- **Keyboard Navigation**: Complete keyboard accessibility
- **Auto-play Options**: Configurable automatic progression
- **Lazy Loading**: Optimized image loading for performance

## Component API

### Props

```typescript
interface CarouselProps {
  /** Carousel items */
  items: CarouselItem[];
  
  /** Display configuration */
  config?: CarouselConfig;
  
  /** Event handlers */
  onSlideChange?: (currentIndex: number) => void;
  onItemClick?: (item: CarouselItem, index: number) => void;
  
  /** Accessibility */
  ariaLabel?: string;
  
  /** Styling */
  className?: string;
  theme?: 'light' | 'dark';
}

interface CarouselItem {
  id: string;
  type: 'image' | 'video' | 'content';
  src?: string;
  alt?: string;
  title?: string;
  description?: string;
  content?: React.ReactNode;
  thumbnail?: string;
  metadata?: Record<string, any>;
}

interface CarouselConfig {
  /** Number of visible items */
  itemsPerView?: number | 'auto';
  
  /** Spacing between items */
  spacing?: number;
  
  /** Auto-play settings */
  autoPlay?: {
    enabled: boolean;
    interval: number;
    pauseOnHover: boolean;
  };
  
  /** Navigation controls */
  navigation?: {
    arrows: boolean;
    dots: boolean;
    thumbnails: boolean;
  };
  
  /** Interaction options */
  loop?: boolean;
  draggable?: boolean;
  zoom?: boolean;
  
  /** Responsive breakpoints */
  breakpoints?: {
    [key: number]: Partial<CarouselConfig>;
  };
}
```

## Advanced Features

### Touch Gestures Implementation

```jsx
const useTouchGestures = ({ onSwipeLeft, onSwipeRight, threshold = 50 }) => {
  const [touchStart, setTouchStart] = useState(null);
  const [touchEnd, setTouchEnd] = useState(null);
  
  const handleTouchStart = (e) => {
    setTouchEnd(null);
    setTouchStart(e.targetTouches[0].clientX);
  };
  
  const handleTouchMove = (e) => {
    setTouchEnd(e.targetTouches[0].clientX);
  };
  
  const handleTouchEnd = () => {
    if (!touchStart || !touchEnd) return;
    
    const distance = touchStart - touchEnd;
    const isLeftSwipe = distance > threshold;
    const isRightSwipe = distance < -threshold;
    
    if (isLeftSwipe) onSwipeLeft();
    if (isRightSwipe) onSwipeRight();
  };
  
  return {
    onTouchStart: handleTouchStart,
    onTouchMove: handleTouchMove,
    onTouchEnd: handleTouchEnd
  };
};
```

### Auto-play with Pause Controls

```jsx
const useAutoPlay = ({ enabled, interval, pauseOnHover }, { next, isPlaying }) => {
  const [isPaused, setIsPaused] = useState(false);
  const intervalRef = useRef(null);
  
  useEffect(() => {
    if (!enabled || isPaused || !isPlaying) {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
      return;
    }
    
    intervalRef.current = setInterval(() => {
      next();
    }, interval);
    
    return () => {
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, [enabled, interval, isPaused, isPlaying, next]);
  
  const pause = () => setIsPaused(true);
  const resume = () => setIsPaused(false);
  
  const hoverHandlers = pauseOnHover ? {
    onMouseEnter: pause,
    onMouseLeave: resume
  } : {};
  
  return { isPaused, pause, resume, hoverHandlers };
};
```

## Zoom Functionality

### Image Zoom Implementation

```jsx
const useImageZoom = () => {
  const [zoomedImage, setZoomedImage] = useState(null);
  const [zoomLevel, setZoomLevel] = useState(1);
  const [zoomPosition, setZoomPosition] = useState({ x: 0, y: 0 });
  
  const openZoom = (image) => {
    setZoomedImage(image);
    document.body.style.overflow = 'hidden';
  };
  
  const closeZoom = () => {
    setZoomedImage(null);
    setZoomLevel(1);
    setZoomPosition({ x: 0, y: 0 });
    document.body.style.overflow = 'auto';
  };
  
  const handleZoomIn = () => {
    setZoomLevel(prev => Math.min(prev * 1.5, 4));
  };
  
  const handleZoomOut = () => {
    setZoomLevel(prev => Math.max(prev / 1.5, 1));
  };
  
  const handleMouseMove = (e) => {
    if (zoomLevel > 1) {
      const rect = e.currentTarget.getBoundingClientRect();
      const x = ((e.clientX - rect.left) / rect.width - 0.5) * 100;
      const y = ((e.clientY - rect.top) / rect.height - 0.5) * 100;
      setZoomPosition({ x, y });
    }
  };
  
  return {
    zoomedImage,
    zoomLevel,
    zoomPosition,
    openZoom,
    closeZoom,
    handleZoomIn,
    handleZoomOut,
    handleMouseMove
  };
};

const ZoomModal = ({ image, zoomLevel, zoomPosition, onClose, onZoomIn, onZoomOut, onMouseMove }) => (
```

## Responsive Behavior

### Breakpoint Management

```jsx
const useResponsiveConfig = (config) => {
  const [currentConfig, setCurrentConfig] = useState(config);
  
  useEffect(() => {
    const updateConfig = () => {
      const width = window.innerWidth;
      const breakpoints = config.breakpoints || {};
      
      // Find the largest breakpoint that's smaller than current width
      const activeBreakpoint = Object.keys(breakpoints)
        .map(Number)
        .sort((a, b) => b - a)
        .find(bp => width >= bp);
      
      if (activeBreakpoint && breakpoints[activeBreakpoint]) {
        setCurrentConfig({
          ...config,
          ...breakpoints[activeBreakpoint]
        });
      } else {
        setCurrentConfig(config);
      }
    };
    
    updateConfig();
    window.addEventListener('resize', updateConfig);
    
    return () => window.removeEventListener('resize', updateConfig);
  }, [config]);
  
  return currentConfig;
};
```

## Accessibility Features

### Keyboard Navigation

```jsx
const useKeyboardNavigation = ({ onPrevious, onNext, onSelect }) => {
  useEffect(() => {
    const handleKeyDown = (e) => {
      switch (e.key) {
        case 'ArrowLeft':
          e.preventDefault();
          onPrevious();
          break;
        case 'ArrowRight':
          e.preventDefault();
          onNext();
          break;
        case 'Enter':
        case ' ':
          e.preventDefault();
          onSelect();
          break;
        case 'Escape':
          // Close zoom or exit fullscreen
          break;
      }
    };
    
    document.addEventListener('keydown', handleKeyDown);
    return () => document.removeEventListener('keydown', handleKeyDown);
  }, [onPrevious, onNext, onSelect]);
};
```

### Screen Reader Support

```jsx
const CarouselWithA11y = ({ items, current, ...props }) => (
  <div
    className="carousel"
    role="region"
    aria-label={props.ariaLabel || "Image carousel"}
    aria-live="polite"
  >
    <div className="carousel-track" aria-atomic="true">
      {items.map((item, index) => (
        <div
          key={item.id}
          className="carousel-item"
          role="group"
          aria-roledescription="slide"
          aria-label={`${index + 1} of ${items.length}`}
          aria-hidden={index !== current}
        >
          {/* Item content */}
        </div>
      ))}
    </div>
    
    {/* Announce current slide to screen readers */}
    <div className="sr-only" aria-live="polite">
      Slide {current + 1} of {items.length}
      {items[current].title && `: ${items[current].title}`}
    </div>
  </div>
);
```
