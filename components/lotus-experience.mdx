# LÖTUS Experience Component

The LÖTUS Experience section communicates our brand values and exclusive offerings through carefully crafted content and visual service cards. This component embodies our commitment to wellness excellence and natural harmony.

## Overview

This section presents the essence of LÖTUS through:
- **Brand Narrative**: Compelling copy emphasizing our core values
- **Service Cards**: Visual representation of exclusive offerings
- **Call-to-Action**: "DISCOVER" button leading to detailed information

## Component API

### Props

```typescript
interface LotusExperienceProps {
  /** Main section title */
  title?: string;
  
  /** Experience description text */
  description?: string;
  
  /** Array of service offerings */
  services: ServiceCard[];
  
  /** Call-to-action configuration */
  cta?: {
    text: string;
    href: string;
    variant?: 'primary' | 'secondary';
  };
  
  /** Section styling options */
  variant?: 'light' | 'dark' | 'natural';
  
  /** Animation preferences */
  animationEnabled?: boolean;
}

interface ServiceCard {
  id: string;
  title: string;
  description: string;
  icon: string;
  image?: string;
  features?: string[];
  badge?: string;
}
```

## Service Cards Design

### Card Layout Structure

```jsx
const ServiceCard = ({ service }: { service: ServiceCard }) => (
  <div className="service-card">
    <div className="card-header">
      <div className="card-icon">
        <Icon name={service.icon} size="large" />
      </div>
      {service.badge && (
        <span className="card-badge">{service.badge}</span>
      )}
    </div>
    
    <div className="card-content">
      <h3 className="card-title">{service.title}</h3>
      <p className="card-description">{service.description}</p>
      
      {service.features && (
        <ul className="card-features">
          {service.features.map((feature, index) => (
            <li key={index}>{feature}</li>
          ))}
        </ul>
      )}
    </div>
    
    {service.image && (
      <div className="card-image">
        <Image
          src={service.image}
          alt={service.title}
          width={300}
          height={200}
          style={{ objectFit: 'cover' }}
        />
      </div>
    )}
  </div>
);
```

### Animation Implementation

```jsx
import { motion } from 'framer-motion';

const containerVariants = {
  hidden: { opacity: 0 },
  visible: {
    opacity: 1,
    transition: {
      staggerChildren: 0.2,
      delayChildren: 0.1
    }
  }
};

const cardVariants = {
  hidden: { 
    opacity: 0, 
    y: 50 
  },
  visible: { 
    opacity: 1, 
    y: 0,
    transition: {
      duration: 0.6,
      ease: "easeOut"
    }
  }
};

const AnimatedLotusExperience = ({ services, ...props }) => (
  <motion.section
    className="lotus-experience"
    variants={containerVariants}
    initial="hidden"
    whileInView="visible"
    viewport={{ once: true, margin: "-100px" }}
  >
    <motion.div className="service-grid">
      {services.map((service) => (
        <motion.div
          key={service.id}
          variants={cardVariants}
          className="service-card"
          whileHover={{ scale: 1.02 }}
        >
          <ServiceCard service={service} />
        </motion.div>
      ))}
    </motion.div>
  </motion.section>
);
```

### Keyboard Navigation

```typescript
const handleKeyDown = (event: KeyboardEvent, serviceId: string) => {
  switch (event.key) {
    case 'Enter':
    case ' ':
      event.preventDefault();
      navigateToService(serviceId);
      break;
    case 'ArrowRight':
      focusNextCard();
      break;
    case 'ArrowLeft':
      focusPreviousCard();
      break;
  }
};
```
