# Next Event Component

The Next Event component presents upcoming retreat opportunities with immersive villa photo carousels, detailed information, and integrated booking functionality. It serves as the primary conversion point on the landing page.

## Overview

This component dynamically displays:
- **Event Information**: Dates, location, and retreat details
- **Villa Carousels**: High-quality photo galleries of retreat locations
- **Availability Status**: Real-time booking availability
- **Booking Integration**: Direct connection to reservation system

## Villa Carousel Implementation

### Carousel Component Structure

```jsx
const VillaCarousel = ({ images, config }: VillaCarouselProps) => {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [zoomedImage, setZoomedImage] = useState(null);
  
  return (
    <div className="villa-carousel">
      <div className="carousel-main">
        <Image
          src={images[currentIndex].src}
          alt={images[currentIndex].alt}
          width={800}
          height={600}
          style={{ objectFit: 'cover' }}
          onClick={() => config.enableZoom && setZoomedImage(images[currentIndex])}
        />
        
        <div className="carousel-controls">
          <button 
            onClick={() => navigateCarousel('prev')}
            aria-label="Previous image"
          >
            <ChevronLeft />
          </button>
          <button 
            onClick={() => navigateCarousel('next')}
            aria-label="Next image"
          >
            <ChevronRight />
          </button>
        </div>
        
        {images[currentIndex].caption && (
          <div className="image-caption">
            {images[currentIndex].caption}
          </div>
        )}
      </div>
      
      {config.showThumbnails && (
        <div className="carousel-thumbnails">
          {images.map((image, index) => (
            <button
              key={image.id}
              className={`thumbnail ${index === currentIndex ? 'active' : ''}`}
              onClick={() => setCurrentIndex(index)}
            >
              <Image
                src={image.src}
                alt={image.alt}
                width={100}
                height={75}
                style={{ objectFit: 'cover' }}
              />
            </button>
          ))}
        </div>
      )}
      
      {zoomedImage && (
        <ImageZoomModal
          image={zoomedImage}
          onClose={() => setZoomedImage(null)}
        />
      )}
    </div>
  );
};
```

## Booking Integration

### CTA Button Component

```jsx
const BookingCTA = ({ event, booking }: BookingCTAProps) => {
  const [isLoading, setIsLoading] = useState(false);
  
  const handleBookingClick = async () => {
    if (event.status === 'sold-out') return;
    
    setIsLoading(true);
    
    try {
      // Track booking intent
      analytics.track('booking_cta_clicked', {
        event_id: event.id,
        location: event.location.city,
        dates: event.dates
      });
      
      // Redirect to booking flow
      router.push(`${booking.redirectUrl}?event=${event.id}`);
    } catch (error) {
      console.error('Booking navigation error:', error);
    } finally {
      setIsLoading(false);
    }
  };
  
  const getButtonText = () => {
    if (isLoading) return 'Loading...';
    if (event.status === 'sold-out') return 'SOLD OUT';
    return booking.ctaText;
  };
  
  return (
    <button
      className={`booking-cta booking-cta--${event.status}`}
      onClick={handleBookingClick}
      disabled={event.status === 'sold-out' || isLoading}
      aria-label={`Book ${event.title} retreat`}
    >
      {getButtonText()}
      {!isLoading && event.status !== 'sold-out' && (
        <ArrowRight className="cta-icon" />
      )}
    </button>
  );
};
```

### Real-time Availability Updates

```jsx
const useRealtimeAvailability = (eventId: string) => {
  const [availability, setAvailability] = useState(null);
  
  useEffect(() => {
    const ws = new WebSocket(`${WS_URL}/availability/${eventId}`);
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      setAvailability(data.availability);
    };
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
    
    return () => ws.close();
  }, [eventId]);
  
  return availability;
};
```

### Touch Gestures

```jsx
const useTouchGestures = (onSwipeLeft, onSwipeRight) => {
  const [touchStart, setTouchStart] = useState(null);
  const [touchEnd, setTouchEnd] = useState(null);
  
  const minSwipeDistance = 50;
  
  const onTouchStart = (e) => {
    setTouchEnd(null);
    setTouchStart(e.targetTouches[0].clientX);
  };
  
  const onTouchMove = (e) => {
    setTouchEnd(e.targetTouches[0].clientX);
  };
  
  const onTouchEnd = () => {
    if (!touchStart || !touchEnd) return;
    
    const distance = touchStart - touchEnd;
    const isLeftSwipe = distance > minSwipeDistance;
    const isRightSwipe = distance < -minSwipeDistance;
    
    if (isLeftSwipe) onSwipeLeft();
    if (isRightSwipe) onSwipeRight();
  };
  
  return { onTouchStart, onTouchMove, onTouchEnd };
};
```

## Performance Optimization

### Image Loading Strategy

```jsx
const OptimizedVillaImages = ({ images }) => {
  const [loadedImages, setLoadedImages] = useState(new Set());
  
  const handleImageLoad = (imageId) => {
    setLoadedImages(prev => new Set([...prev, imageId]));
  };
  
  return (
    <>
      {/* Preload featured image */}
      <link
        rel="preload"
        as="image"
        href={images.find(img => img.featured)?.src}
      />
      
      {images.map((image, index) => (
        <Image
          key={image.id}
          src={image.src}
          alt={image.alt}
          width={800}
          height={600}
          priority={index === 0}
          loading={index > 2 ? 'lazy' : 'eager'}
          onLoad={() => handleImageLoad(image.id)}
          className={loadedImages.has(image.id) ? 'loaded' : 'loading'}
        />
      ))}
    </>
  );
};
```
