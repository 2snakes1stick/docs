# Booking Widget Component

The Booking Widget is a sophisticated component that handles the entire reservation process, from initial inquiry to payment confirmation. It provides a seamless user experience with real-time availability checks and secure payment processing.

## Overview

The booking widget manages:
- **Availability Checking**: Real-time retreat availability verification
- **Guest Information**: Comprehensive guest details and preferences
- **Payment Processing**: Secure payment handling with multiple options
- **Confirmation Flow**: Booking confirmation and receipt generation

## Component API

### Props

```typescript
interface BookingWidgetProps {
  /** Pre-selected retreat event */
  eventId?: string;
  
  /** Widget display mode */
  mode?: 'inline' | 'modal' | 'fullscreen';
  
  /** Booking flow configuration */
  config?: BookingConfig;
  
  /** Event handlers */
  onBookingComplete?: (booking: Booking) => void;
  onBookingError?: (error: BookingError) => void;
  onStepChange?: (step: BookingStep) => void;
  
  /** UI customization */
  theme?: 'light' | 'dark' | 'natural';
  className?: string;
}

interface BookingConfig {
  /** Available payment methods */
  paymentMethods: PaymentMethod[];
  
  /** Required guest information fields */
  requiredFields: GuestInfoField[];
  
  /** Booking terms and conditions */
  termsUrl: string;
  
  /** Cancellation policy */
  cancellationPolicy: string;
  
  /** Currency for pricing */
  currency: string;
  
  /** Tax configuration */
  taxConfig?: TaxConfig;
}

interface Booking {
  id: string;
  eventId: string;
  guestInfo: GuestInfo;
  paymentInfo: PaymentInfo;
  totalAmount: number;
  status: BookingStatus;
  confirmationCode: string;
  createdAt: string;
}
```

## Payment Integration

### Stripe Integration

```jsx
import { Elements } from '@stripe/react-stripe-js';
import { loadStripe } from '@stripe/stripe-js';

const stripePromise = loadStripe(process.env.NEXT_PUBLIC_STRIPE_PUBLISHABLE_KEY);

const StripePaymentForm = ({ amount, currency, onSuccess, onError }) => {
  const stripe = useStripe();
  const elements = useElements();
  
  const handleSubmit = async (event) => {
    event.preventDefault();
    
    if (!stripe || !elements) return;
    
    const { error, paymentMethod } = await stripe.createPaymentMethod({
      type: 'card',
      card: elements.getElement(CardElement),
    });
    
    if (error) {
      onError(error);
      return;
    }
    
    // Create payment intent on server
    const { clientSecret } = await createPaymentIntent({
      amount: amount * 100, // Convert to cents
      currency,
      paymentMethodId: paymentMethod.id
    });
    
    const { error: confirmError } = await stripe.confirmCardPayment(clientSecret);
    
    if (confirmError) {
      onError(confirmError);
    } else {
      onSuccess({ paymentMethod, clientSecret });
    }
  };
  
  return (
    <form onSubmit={handleSubmit} className="stripe-form">
      <CardElement
        options={{
          style: {
            base: {
              fontSize: '16px',
              color: '#424770',
              '::placeholder': {
                color: '#aab7c4',
              },
            },
          },
        }}
      />
      
      <button
        type="submit"
        disabled={!stripe || isProcessing}
        className="payment-submit-btn"
      >
        {isProcessing ? 'Processing...' : `Pay ${formatCurrency(amount, currency)}`}
      </button>
    </form>
  );
};
```

## Real-time Features

### Availability Checking

```jsx
const useRealTimeAvailability = (eventId) => {
  const [availability, setAvailability] = useState(null);
  const [isChecking, setIsChecking] = useState(false);
  
  const checkAvailability = useCallback(async () => {
    setIsChecking(true);
    try {
      const response = await fetch(`/api/events/${eventId}/availability`);
      const data = await response.json();
      setAvailability(data);
    } catch (error) {
      console.error('Availability check failed:', error);
    } finally {
      setIsChecking(false);
    }
  }, [eventId]);
  
  useEffect(() => {
    if (eventId) {
      checkAvailability();
      
      // Set up real-time updates
      const ws = new WebSocket(`${WS_URL}/availability/${eventId}`);
      ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        setAvailability(data.availability);
      };
      
      return () => ws.close();
    }
  }, [eventId, checkAvailability]);
  
  return { availability, isChecking, checkAvailability };
};
```

### Booking Timer

```jsx
const BookingTimer = ({ duration = 900, onExpire }) => { // 15 minutes
  const [timeLeft, setTimeLeft] = useState(duration);
  
  useEffect(() => {
    if (timeLeft <= 0) {
      onExpire();
      return;
    }
    
    const timer = setTimeout(() => {
      setTimeLeft(timeLeft - 1);
    }, 1000);
    
    return () => clearTimeout(timer);
  }, [timeLeft, onExpire]);
  
  const formatTime = (seconds) => {
    const minutes = Math.floor(seconds / 60);
    const remainingSeconds = seconds % 60;
    return `${minutes}:${remainingSeconds.toString().padStart(2, '0')}`;
  };
  
  return (
    <div className={`booking-timer ${timeLeft < 60 ? 'urgent' : ''}`}>
      <Clock className="timer-icon" />
      <span>Time remaining: {formatTime(timeLeft)}</span>
    </div>
  );
};
```

### Custom Validation Hooks

```jsx
const useFormValidation = (schema) => {
  const [errors, setErrors] = useState({});
  
  const validate = (data) => {
    try {
      schema.parse(data);
      setErrors({});
      return true;
    } catch (error) {
      if (error instanceof z.ZodError) {
        const fieldErrors = {};
        error.errors.forEach((err) => {
          fieldErrors[err.path.join('.')] = err.message;
        });
        setErrors(fieldErrors);
      }
      return false;
    }
  };
  
  const getFieldError = (fieldName) => errors[fieldName];
  
  return { validate, getFieldError, errors };
};
```

## Error Handling

### Error Boundary

```jsx
class BookingErrorBoundary extends Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }
  
  componentDidCatch(error, errorInfo) {
    console.error('Booking widget error:', error, errorInfo);
    
    // Send error to monitoring service
    analytics.track('booking_error', {
      error: error.message,
      errorInfo,
      userId: this.props.userId
    });
  }
  
  render() {
    if (this.state.hasError) {
      return (
        <div className="booking-error">
          <h2>Something went wrong</h2>
          <p>We're sorry, but there was an error processing your booking.</p>
          <button
            onClick={() => this.setState({ hasError: false, error: null })}
          >
            Try Again
          </button>
        </div>
      );
    }
    
    return this.props.children;
  }
}
```

### Error Recovery

```jsx
const useBookingRecovery = () => {
  const [savedBooking, setSavedBooking] = useState(null);
  
  const saveBookingState = (bookingData) => {
    localStorage.setItem('lotus_booking_draft', JSON.stringify({
      ...bookingData,
      timestamp: Date.now()
    }));
  };
  
  const recoverBookingState = () => {
    const saved = localStorage.getItem('lotus_booking_draft');
    if (saved) {
      const booking = JSON.parse(saved);
      // Check if booking is not too old (e.g., 1 hour)
      if (Date.now() - booking.timestamp < 3600000) {
        setSavedBooking(booking);
        return booking;
      }
    }
    return null;
  };
  
  const clearSavedBooking = () => {
    localStorage.removeItem('lotus_booking_draft');
    setSavedBooking(null);
  };
  
  return { saveBookingState, recoverBookingState, clearSavedBooking, savedBooking };
};
```
